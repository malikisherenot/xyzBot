cmake_minimum_required(VERSION 3.21)

project(xyzBot VERSION 2.1.0)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set(CMAKE_OSX_ARCHITECTURES "arm64")
else()
  set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64")
endif()

if(NOT DEFINED ENV{GEODE_SDK})
  message(FATAL_ERROR "Unable to find Geode SDK! Set the GEODE_SDK environment variable.")
endif()

file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(${PROJECT_NAME} SHARED ${SOURCES})

# Video rendering uses FFmpeg's own libraries (libav*), linked statically from ffmpeg/<platform>/{include,lib}.
# Build a small set with ffmpeg/build-ffmpeg.sh. Without the libraries the mod still builds, rendering is just off.
option(XYZ_MEDIACODEC "Android hardware H.264 encoder (needs FFmpeg built with --enable-jni --enable-mediacodec)" OFF)
if(ANDROID)
  set(XYZ_FF_PLAT "android-${ANDROID_ABI}")
elseif(WIN32)
  set(XYZ_FF_PLAT "windows-x64")
else()
  set(XYZ_FF_PLAT "")
endif()
set(XYZ_FF_DIR "${CMAKE_CURRENT_SOURCE_DIR}/ffmpeg/${XYZ_FF_PLAT}")

# finds one FFmpeg library by its name, whatever the build called the file
function(xyz_find_ff name out)
  foreach(cand "${XYZ_FF_DIR}/lib/${name}.lib" "${XYZ_FF_DIR}/lib/lib${name}.a" "${XYZ_FF_DIR}/lib/lib${name}.lib" "${XYZ_FF_DIR}/lib/${name}.a")
    if(EXISTS "${cand}")
      set(${out} "${cand}" PARENT_SCOPE)
      return()
    endif()
  endforeach()
  set(${out} "" PARENT_SCOPE)
endfunction()

set(XYZ_FF_LIBS "")
set(XYZ_FF_OK FALSE)
if(XYZ_FF_PLAT AND EXISTS "${XYZ_FF_DIR}/include/libavcodec/avcodec.h")
  set(XYZ_FF_OK TRUE)
  # dependents first, so the static linker finds everything
  foreach(n avformat avcodec swresample swscale avutil)
    xyz_find_ff(${n} f)
    if(f STREQUAL "")
      file(GLOB present "${XYZ_FF_DIR}/lib/*")
      message(WARNING "xyzBot: FFmpeg library '${n}' not found in ${XYZ_FF_DIR}/lib (found: ${present})")
      set(XYZ_FF_OK FALSE)
    else()
      list(APPEND XYZ_FF_LIBS "${f}")
    endif()
  endforeach()
endif()

if(XYZ_FF_OK)
  message(STATUS "xyzBot: rendering with FFmpeg from ${XYZ_FF_DIR}")
  target_compile_definitions(${PROJECT_NAME} PRIVATE XYZ_LIBAV=1)
  target_include_directories(${PROJECT_NAME} PRIVATE "${XYZ_FF_DIR}/include")
  target_link_libraries(${PROJECT_NAME} ${XYZ_FF_LIBS})
  if(WIN32)
    target_link_libraries(${PROJECT_NAME} bcrypt)
  endif()
  if(ANDROID)
    target_link_libraries(${PROJECT_NAME} log z)
    # keep the binary small: drop unused code and do not re-export FFmpeg's symbols
    target_compile_options(${PROJECT_NAME} PRIVATE -ffunction-sections -fdata-sections)
    target_link_options(${PROJECT_NAME} PRIVATE -Wl,--gc-sections -Wl,--exclude-libs,ALL)
    if(XYZ_MEDIACODEC)
      target_compile_definitions(${PROJECT_NAME} PRIVATE XYZ_MEDIACODEC=1)
      target_link_libraries(${PROJECT_NAME} mediandk android)
    endif()
  endif()
else()
  message(WARNING "xyzBot: no usable FFmpeg libraries in ffmpeg/${XYZ_FF_PLAT}, video rendering is disabled in this build")
  target_compile_definitions(${PROJECT_NAME} PRIVATE XYZ_LIBAV=0)
endif()
if(NOT XYZ_MEDIACODEC)
  target_compile_definitions(${PROJECT_NAME} PRIVATE XYZ_MEDIACODEC=0)
endif()

add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
setup_geode_mod(${PROJECT_NAME})

# Lua 5.4 for the scripting (src/core/lua), built from the official source as a small static library
CPMAddPackage(NAME lua GITHUB_REPOSITORY lua/lua GIT_TAG v5.4.7 DOWNLOAD_ONLY YES)
if(lua_ADDED)
  file(GLOB XYZ_LUA_SRC "${lua_SOURCE_DIR}/*.c")
  list(FILTER XYZ_LUA_SRC EXCLUDE REGEX "/(lua|luac|onelua|ltests)\\.c$")
  add_library(xyz_lua STATIC ${XYZ_LUA_SRC})
  set_target_properties(xyz_lua PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
  target_include_directories(xyz_lua PUBLIC "${lua_SOURCE_DIR}")
  target_link_libraries(${PROJECT_NAME} xyz_lua)
endif()
